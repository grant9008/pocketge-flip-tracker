import com.pocketge.tracker.GeOfferPriceOverlay;

/**
 * What the plugin draws ON the Grand Exchange, as an instructional diagram.
 *
 * The overlay chrome here is REAL: every ring, chip and border is painted by
 * GeOfferPriceOverlay and GeOfferGridOverlay themselves, given widget bounds
 * the way the live client gives them. What is drawn underneath — the Exchange
 * window, the offer rows, the chatbox — is a plain illustration of Jagex's
 * interface, not a capture of it, because no game client runs here.
 *
 * That split is the point, and the caption in the README says so: the parts
 * that are the plugin are exact, and the backdrop is there to show where they
 * land. It is a diagram, and it should not be passed off as a screenshot.
 */
public class GeDiagram
{
	private static final Color GAME_BG = new Color(0x3E, 0x37, 0x2C);
	private static final Color GAME_PANEL = new Color(0x2E, 0x29, 0x21);
	private static final Color GAME_EDGE = new Color(0x5A, 0x50, 0x40);
	private static final Color GAME_TEXT = new Color(0xC8, 0xBE, 0xA8);
	private static final Color CAPTION = new Color(0xE5, 0xC1, 0x58);
	private static final Color SHEET_BG = new Color(0x1B, 0x18, 0x15);

	// ---------------------------------------------------------------- stubs

	private static final class FakeWidget implements Widget
	{
		String text = "";
		boolean hidden;
		Rectangle bounds = new Rectangle();
		String[] actions;
		final List<Widget> kids = new ArrayList<>();

		FakeWidget(String text, Rectangle bounds)
		{
			this.text = text;
			this.bounds = bounds;
		}

		public String getText() { return text; }
		public boolean isHidden() { return hidden; }
		public Rectangle getBounds() { return bounds; }
		public String[] getActions() { return actions; }
		public Widget[] getStaticChildren() { return kids.toArray(new Widget[0]); }
		public Widget[] getDynamicChildren() { return null; }
		public Widget[] getNestedChildren() { return null; }
	}

	private static final class FakeClient implements Client
	{
		final Map<Integer, Widget> widgets = new HashMap<>();
		int w = 360, h = 270;
		Point mouse;

		public Widget getWidget(int id) { return widgets.get(id); }
		public int getCanvasWidth() { return w; }
		public int getCanvasHeight() { return h; }
		public Point getMouseCanvasPosition() { return mouse; }
	}

	private static GeOfferPriceOverlay priceOverlay(Client c) throws Exception
	{
		final Constructor<?> k = GeOfferPriceOverlay.class.getDeclaredConstructors()[0];
		k.setAccessible(true);
		return (GeOfferPriceOverlay) k.newInstance(c, new net.runelite.client.game.ItemManager());
	}

	private static GeOfferGridOverlay gridOverlay(Client c) throws Exception
	{
		final Constructor<?> k = GeOfferGridOverlay.class.getDeclaredConstructors()[0];
		k.setAccessible(true);
		return (GeOfferGridOverlay) k.newInstance(c,
			new net.runelite.client.ui.overlay.tooltip.TooltipManager());
	}

	// ------------------------------------------------------------- painting

	private static void hints(Graphics2D g)
	{
		g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
		g.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING, RenderingHints.VALUE_TEXT_ANTIALIAS_ON);
		g.setRenderingHint(RenderingHints.KEY_FRACTIONALMETRICS, RenderingHints.VALUE_FRACTIONALMETRICS_OFF);
	}

	private static void box(Graphics2D g, Rectangle r, Color fill)
	{
		g.setColor(fill);
		g.fillRect(r.x, r.y, r.width, r.height);
		g.setColor(GAME_EDGE);
		g.setStroke(new BasicStroke(1f));
		g.drawRect(r.x, r.y, r.width - 1, r.height - 1);
	}

	private static void text(Graphics2D g, String s, int x, int y, Color c, int size, boolean bold)
	{
		g.setFont(new Font(Font.SANS_SERIF, bold ? Font.BOLD : Font.PLAIN, size));
		g.setColor(c);
		g.drawString(s, x, y);
	}

	/** The "Set up offer" screen, drawn plainly. Returns the client whose
	 *  widget bounds match what was drawn. */
	private static FakeClient drawSetupScreen(Graphics2D g, int w, int h,
		String prompt, String enteredPrice)
	{
		final FakeClient c = new FakeClient();
		c.w = w;
		c.h = h;
		g.setColor(GAME_BG);
		g.fillRect(0, 0, w, h);

		final Rectangle setupR = new Rectangle(10, 10, w - 20, h - 86);
		box(g, setupR, GAME_PANEL);
		text(g, "Set up offer", 20, 27, GAME_TEXT, 12, true);

		text(g, "Quantity:", 22, 56, GAME_TEXT, 11, false);
		final Rectangle qtyField = new Rectangle(22, 62, 120, 20);
		box(g, qtyField, SHEET_BG);
		text(g, "1", 28, 77, GAME_TEXT, 11, false);

		text(g, "Price per item:", 160, 56, GAME_TEXT, 11, false);
		final Rectangle priceField = new Rectangle(160, 62, 120, 20);
		box(g, priceField, SHEET_BG);
		text(g, enteredPrice == null ? "1,146 coins" : enteredPrice + " coins",
			166, 77, GAME_TEXT, 11, false);

		// The two "..." controls the overlay rings, labelled the way the game
		// labels them so findEntryControl matches on its own terms.
		final Rectangle qtyDots = new Rectangle(112, 92, 26, 18);
		final Rectangle priceDots = new Rectangle(250, 92, 26, 18);
		box(g, qtyDots, GAME_BG);
		text(g, "...", 119, 105, GAME_TEXT, 11, true);
		box(g, priceDots, GAME_BG);
		text(g, "...", 257, 105, GAME_TEXT, 11, true);

		final FakeWidget setup = new FakeWidget("", setupR);
		final FakeWidget qtyBtn = new FakeWidget("", qtyDots);
		qtyBtn.actions = new String[]{"Enter quantity"};
		final FakeWidget priceBtn = new FakeWidget("", priceDots);
		priceBtn.actions = new String[]{"Enter price"};
		setup.kids.add(qtyBtn);
		setup.kids.add(priceBtn);
		if (enteredPrice != null)
		{
			// readSetupPrice looks for a widget whose text mentions coins.
			setup.kids.add(new FakeWidget(enteredPrice + " coins", priceField));
		}
		c.widgets.put(InterfaceID.GeOffers.SETUP, setup);

		// Chatbox strip along the bottom, where the chip is anchored.
		final Rectangle chat = new Rectangle(6, h - 72, w - 12, 66);
		box(g, chat, SHEET_BG);
		final Rectangle mes = new Rectangle(14, h - 34, w - 28, 14);
		if (prompt != null)
		{
			text(g, prompt, 16, h - 22, GAME_TEXT, 11, false);
		}
		c.widgets.put(InterfaceID.Chatbox.CHATAREA, new FakeWidget("", chat));
		c.widgets.put(InterfaceID.Chatbox.MES_TEXT, new FakeWidget(prompt == null ? "" : prompt, mes));
		c.widgets.put(InterfaceID.Chatbox.MES_TEXT2, new FakeWidget("", new Rectangle(14, h - 18, 10, 10)));
		return c;
	}

	/** One panel of the sheet: a scene image plus a numbered caption. */
	private static void panel(Graphics2D g, BufferedImage scene, int x, int y,
		String step, String caption)
	{
		g.drawImage(scene, x, y, null);
		g.setColor(new Color(0x4A, 0x42, 0x36));
		g.setStroke(new BasicStroke(1f));
		g.drawRect(x, y, scene.getWidth() - 1, scene.getHeight() - 1);

		final int ty = y + scene.getHeight() + 18;
		text(g, step, x, ty, CAPTION, 13, true);
		g.setFont(new Font(Font.SANS_SERIF, Font.PLAIN, 12));
		final FontMetrics fm = g.getFontMetrics();
		final int stepW = fm.stringWidth(step) + 8;
		g.setColor(new Color(0xD9, 0xD3, 0xC7));
		// wrap the caption under the panel
		int cx = x + stepW;
		int cy = ty;
		for (String word : caption.split(" "))
		{
			final int ww = fm.stringWidth(word + " ");
			if (cx + ww > x + scene.getWidth())
			{
				cx = x;
				cy += 16;
			}
			g.drawString(word, cx, cy);
			cx += ww;
		}
	}

	public static void main(String[] args) throws Exception
	{
		System.setProperty("awt.useSystemAAFontSettings", "off");
		final File out = new File(args.length > 0 ? args[0] : ".");
		out.mkdirs();

		final int sw = 330, sh = 250;
		final BufferedImage[] scenes = new BufferedImage[4];

		// 1 — nothing open: the ring points at the price control.
		scenes[0] = scene(sw, sh, null, null, 1120, 18000);
		// 2 — the price prompt is open: the chip offers the exact price.
		scenes[1] = scene(sw, sh, "Set a price for each item:", null, 1120, 18000);
		// 3 — price entered: the ring has moved to the quantity control.
		scenes[2] = scene(sw, sh, null, "1,120", 1120, 18000);
		// 4 — the quantity prompt: the chip offers how many.
		scenes[3] = scene(sw, sh, "How many do you wish to buy?", "1,120", 1120, 18000);

		final int pad = 26, capH = 52;
		final int W = pad * 2 + sw * 2 + 24;
		final int H = pad * 2 + (sh + capH) * 2 + 34;
		final BufferedImage sheet = new BufferedImage(W, H, BufferedImage.TYPE_INT_RGB);
		final Graphics2D g = sheet.createGraphics();
		hints(g);
		g.setColor(SHEET_BG);
		g.fillRect(0, 0, W, H);
		text(g, "Placing an offer: the plugin marks what to press next",
			pad, pad + 2, CAPTION, 15, true);

		final int top = pad + 24;
		panel(g, scenes[0], pad, top,
			"1.", "A gold ring marks the price control. Click it to open the price box.");
		panel(g, scenes[1], pad + sw + 24, top,
			"2.", "The chip shows the exact price, in full. Click it and the price is filled in.");
		panel(g, scenes[2], pad, top + sh + capH,
			"3.", "Once the price is in, the ring moves to the quantity control.");
		panel(g, scenes[3], pad + sw + 24, top + sh + capH,
			"4.", "The chip now offers how many: the 4-hour limit for a buy, the stack you hold for a sell.");
		g.dispose();

		final File f = new File(out, "ge-offer.png");
		ImageIO.write(sheet, "png", f);
		System.out.println("  ge-offer.png  " + W + "x" + H + "  " + (f.length() / 1024) + "KB");

		slotSheet(out);
		System.exit(0);
	}

	/**
	 * The coloured border on each live offer, and what a red one says.
	 *
	 * Same split as the sheet above: the borders are painted by
	 * GeOfferGridOverlay itself. The tooltip is the overlay's own text —
	 * pulled back out of the TooltipManager it handed it to — drawn here in a
	 * plain box, because in the game the client draws it and there is no
	 * client here.
	 */
	private static void slotSheet(File out) throws Exception
	{
		final int w = 700, h = 252;
		final BufferedImage img = new BufferedImage(w, h, BufferedImage.TYPE_INT_RGB);
		final Graphics2D g = img.createGraphics();
		hints(g);
		g.setColor(SHEET_BG);
		g.fillRect(0, 0, w, h);
		text(g, "Every live offer, colour-coded \u2014 and a red one tells you the fix",
			24, 28, CAPTION, 15, true);

		final FakeClient c = new FakeClient();
		c.w = w;
		c.h = h;
		final Rectangle window = new Rectangle(24, 46, 400, 150);
		box(g, window, GAME_PANEL);
		text(g, "Grand Exchange", 34, 64, GAME_TEXT, 12, true);

		final String[] names = {"Fur", "Emerald necklace", "Adamantite bar", ""};
		final String[] qty = {"18,000", "18,000", "10,000", ""};
		final String[] price = {"319 coins", "717 coins", "1,890 coins", ""};
		final int[] ids = {InterfaceID.GeOffers.INDEX_0, InterfaceID.GeOffers.INDEX_1,
			InterfaceID.GeOffers.INDEX_2, InterfaceID.GeOffers.INDEX_3};
		Rectangle redBox = null;
		for (int i = 0; i < 4; i++)
		{
			final Rectangle r = new Rectangle(34 + i * 96, 74, 88, 112);
			box(g, r, GAME_BG);
			if (!names[i].isEmpty())
			{
				text(g, i == 2 ? "Sell" : "Buy", r.x + 6, r.y + 16, GAME_TEXT, 10, true);
				text(g, qty[i], r.x + 6, r.y + 46, GAME_TEXT, 10, false);
				String n = names[i];
				if (n.length() > 11) { n = n.substring(0, 10) + "\u2026"; }
				text(g, n, r.x + 6, r.y + 60, GAME_TEXT, 9, false);
				text(g, price[i], r.x + 6, r.y + 104, GAME_TEXT, 9, false);
				c.widgets.put(ids[i], new FakeWidget("", r));
			}
			else
			{
				text(g, "Empty", r.x + 6, r.y + 16, new Color(0x7A, 0x71, 0x60), 10, false);
			}
			if (i == 0) { redBox = r; }
		}

		// Hover the red one so the overlay asks for its tooltip.
		c.mouse = new Point(redBox.x + 40, redBox.y + 50);
		final net.runelite.client.ui.overlay.tooltip.TooltipManager tm =
			new net.runelite.client.ui.overlay.tooltip.TooltipManager();
		final Constructor<?> k = GeOfferGridOverlay.class.getDeclaredConstructors()[0];
		k.setAccessible(true);
		final GeOfferGridOverlay grid = (GeOfferGridOverlay) k.newInstance(c, tm);

		final Map<Integer, GeOfferGridOverlay.SlotView> slots = new HashMap<>();
		slots.put(0, slotView("Fur", true, 138, 18000, true, 319, 340, false, -42_000L));
		slots.put(1, slotView("Emerald necklace", true, 12400, 18000, false, 0, 0, false, 486_000L));
		slots.put(2, slotView("Adamantite bar", false, 10000, 10000, false, 0, 0, false, 542_000L));
		grid.setSlots(slots);
		grid.render(g);

		// The overlay's own words, drawn where the client would draw them.
		final List<net.runelite.client.ui.overlay.tooltip.Tooltip> tips = tm.getTooltips();
		if (!tips.isEmpty())
		{
			drawTooltip(g, tips.get(0).getText(), 446, 74, 230);
		}
		text(g, "Hovering the red offer", 446, 64, CAPTION, 12, true);

		g.setColor(new Color(0xD9, 0xD3, 0xC7));
		g.setFont(new Font(Font.SANS_SERIF, Font.PLAIN, 12));
		g.drawString("Green: priced fine.   Red: the market moved past it.", 24, h - 34);
		g.drawString("Right-click a slot for its chart, or to stop being told to reprice it.", 24, h - 16);
		g.dispose();

		final File f = new File(out, "ge-slots.png");
		ImageIO.write(img, "png", f);
		System.out.println("  ge-slots.png  " + w + "x" + h + "  " + (f.length() / 1024) + "KB");
	}

	private static GeOfferGridOverlay.SlotView slotView(String name, boolean buy, int filled,
		int total, boolean adjust, long offerPrice, long target, boolean noMargin, Long profit)
	{
		final GeOfferGridOverlay.SlotView v = new GeOfferGridOverlay.SlotView();
		v.itemName = name;
		v.buy = buy;
		v.filled = filled;
		v.total = total;
		v.needsAdjust = adjust;
		v.offerPrice = offerPrice;
		v.targetPrice = target;
		v.noMargin = noMargin;
		v.projectedProfit = profit;
		v.filledProfit = profit;
		return v;
	}

	/** Render RuneLite's little markup dialect (&lt;/br&gt;, &lt;col=rrggbb&gt;)
	 *  as plain styled lines. */
	private static void drawTooltip(Graphics2D g, String markup, int x, int y, int wrapW)
	{
		final String[] lines = markup.split("</br>");
		final Font f = new Font(Font.SANS_SERIF, Font.PLAIN, 12);
		g.setFont(f);
		final FontMetrics fm = g.getFontMetrics();
		int maxW = 0;
		for (String l : lines)
		{
			maxW = Math.max(maxW, fm.stringWidth(l.replaceAll("<[^>]*>", "")));
		}
		final int boxW = Math.min(Math.max(maxW + 18, 150), wrapW + 18);
		final int boxH = lines.length * 17 + 14;
		g.setColor(new Color(0x28, 0x24, 0x20));
		g.fillRect(x, y, boxW, boxH);
		g.setColor(new Color(0x5A, 0x50, 0x40));
		g.setStroke(new BasicStroke(1f));
		g.drawRect(x, y, boxW - 1, boxH - 1);

		int ly = y + 20;
		for (String line : lines)
		{
			int lx = x + 9;
			// split into <col=..>..</col> runs, keeping each run's colour
			final java.util.regex.Matcher m = java.util.regex.Pattern
				.compile("<col=([0-9a-fA-F]{6})>(.*?)</col>|([^<]+)").matcher(line);
			while (m.find())
			{
				final String colored = m.group(2);
				final String plain = m.group(3);
				final String piece = colored != null ? colored : plain;
				if (piece == null || piece.isEmpty())
				{
					continue;
				}
				g.setColor(colored != null
					? new Color(Integer.parseInt(m.group(1), 16))
					: new Color(0xD9, 0xD3, 0xC7));
				g.drawString(piece, lx, ly);
				lx += fm.stringWidth(piece);
			}
			ly += 17;
		}
	}

	private static BufferedImage scene(int w, int h, String prompt, String entered,
		long price, long qty) throws Exception
	{
		final BufferedImage img = new BufferedImage(w, h, BufferedImage.TYPE_INT_RGB);
		final Graphics2D g = img.createGraphics();
		hints(g);
		final FakeClient c = drawSetupScreen(g, w, h, prompt, entered);
		final GeOfferPriceOverlay o = priceOverlay(c);
		o.setContext("Sapphire necklace", true, price, price, 49, qty);
		o.render(g);   // the real overlay, painting itself
		g.dispose();
		return img;
	}
}
